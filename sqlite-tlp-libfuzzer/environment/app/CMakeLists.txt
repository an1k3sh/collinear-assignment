cmake_minimum_required(VERSION 3.20)
project(sqlite_tlp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(SQLite3 REQUIRED)
find_package(nlohmann_json 3.11 REQUIRED)
file(GLOB SOURCES CONFIGURE_DEPENDS src/*.cpp)
add_library(tlp STATIC ${SOURCES} harness/runtime.cpp)
target_include_directories(tlp PUBLIC include)
target_link_libraries(tlp PUBLIC SQLite::SQLite3 nlohmann_json::nlohmann_json)
target_compile_options(tlp PRIVATE -Wall -Wextra -Werror -g -O1
    -fsanitize=fuzzer-no-link,address,undefined -fno-sanitize-recover=all)
add_executable(tlp_cli harness/cli.cpp)
target_link_libraries(tlp_cli PRIVATE tlp)
target_compile_options(tlp_cli PRIVATE -g -O1 -fsanitize=address,undefined)
target_link_options(tlp_cli PRIVATE -fsanitize=fuzzer-no-link,address,undefined)
add_executable(tlp_fuzz harness/fuzz.cpp)
target_link_libraries(tlp_fuzz PRIVATE tlp)
target_compile_options(tlp_fuzz PRIVATE -g -O1 -fsanitize=fuzzer,address,undefined)
target_link_options(tlp_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
